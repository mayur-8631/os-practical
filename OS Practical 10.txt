from multiprocessing import Process, Pipe


def child_process(connection):

    # Receive message from parent
    message = connection.recv()

    print("Child received:", message)

    connection.close()


# Create pipe
parent_conn, child_conn = Pipe()

# Create child process
p = Process(target=child_process, args=(child_conn,))

# Start child process
p.start()

# Parent sends message
parent_conn.send("Hello from Parent Process")

print("Parent sent: Hello from Parent Process")

# Wait for child process
p.join()

# Close pipe
parent_conn.close()

Task 2: IPC Using Shared Memory
from multiprocessing import Process, Pipe


def sender(conn):
    message = "Hello from Parent Process"
    conn.send(message)
    conn.close()


def receiver(conn):
    message = conn.recv()
    print("Received message:", message)
    conn.close()


if __name__ == "__main__":

    parent_conn, child_conn = Pipe()

    p1 = Process(target=sender, args=(parent_conn,))
    p2 = Process(target=receiver, args=(child_conn,))

    p1.start()
    p2.start()

    p1.join()
    p2.join()
